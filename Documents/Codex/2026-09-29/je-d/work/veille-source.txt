<section id="duel">
  <h2>Mali face au Sénégal</h2>
  <p class="sub">Les deux marchés ne jouent pas le même jeu. Ce qui marche à Bamako ne se transpose pas tel quel à Dakar.</p>
  <div class="compare">
    <div>
      <h3 class="ml">Mali</h3>
      <ul>
        <li>Marché jeune et peu structuré. Aucun classement professionnel sérieux n'existe : les « top 10 » sont écrits par les agences elles-mêmes.</li>
        <li>Les trois leaders n'affichent aucun prix. Le client doit appeler pour savoir combien ça coûte.</li>
        <li>Ticket moyen d'un site : 150 000 à 800 000 FCFA. Le haut du marché plafonne autour de 1,2 M.</li>
        <li>Clientèle : PME, commerces, ONG. Peu de grands comptes digitalisés.</li>
        <li>Acquisition dominée par le référencement et WhatsApp. L'achat média pour soi est quasi inexistant.</li>
        <li>Aucun acteur ne vend de produit ni de SaaS. On facture du temps.</li>
        <li><b>Conséquence</b> : un entrant qui affiche ses prix et vend du récurrent se distingue immédiatement.</li>
      </ul>
    </div>
    <div>
      <h3 class="sn">Sénégal</h3>
      <ul>
        <li>Marché mature. Classements professionnels, annuaires internationaux, chiffres d'affaires et effectifs déclarés publiquement.</li>
        <li>Les prix sont affichés, y compris les abonnements mensuels de community management. Le marché est transparent.</li>
        <li>Ticket moyen d'un site : 300 000 à 1,5 M FCFA. Les portails institutionnels montent à 20 M.</li>
        <li>Clientèle : banques, télécoms, État, bailleurs, multinationales. People Input travaille pour la BCEAO et la BRVM.</li>
        <li>Acquisition par la référence institutionnelle et l'appel d'offres plus que par Google.</li>
        <li>L'intégration Mobile Money, Wave et Orange Money est une compétence vendue en tant que telle.</li>
        <li><b>Conséquence</b> : impossible d'y entrer par le prix. Il faut une spécialité technique ou une référence qui ouvre les portes.</li>
      </ul>
    </div>
  </div>
  <div class="method" style="margin-top:20px">
    <p><b>Le pont entre les deux.</b> Sabma Digital est le seul acteur présent des deux côtés, et elle se déplace : son site garde une page Mali mais son siège, son numéro et son activité sont désormais à Dakar. Le mouvement va du Mali vers le Sénégal, jamais l'inverse.</p>
    <p>Un site vendu 400 000 FCFA à Bamako se vend 800 000 à Dakar pour un travail comparable. L'écart n'est pas dans la compétence, il est dans le pouvoir d'achat des clients.</p>
  </div>

let pays = "mali", classees = [], etat = {};

function trier(p){ return [...DATA[p].agences].sort((a,b)=>
  b.notes.reduce((s,n)=>s+n,0) - a.notes.reduce((s,n)=>s+n,0)); }
const total = a => a.notes.reduce((s,n)=>s+n,0);

function rendreSim(){
  CHAMPS.forEach(c=>{
    document.getElementById("s-"+c.k).value = etat[c.k];
    document.getElementById("v-"+c.k).textContent = F(etat[c.k]) + c.suf;
  });
  const recur = etat.retainers*etat.ticketRet,
        web   = etat.sites*etat.ticketSite,
        spot  = etat.spots*etat.ticketSpot,
        mois  = recur+web+spot,
        clients = etat.retainers + etat.sites + etat.spots;
  document.getElementById("o-mois").textContent  = M(mois)+" FCFA";
  document.getElementById("o-an").textContent    = M(mois*12)+" FCFA";
  document.getElementById("o-recur").textContent = mois ? Math.round(recur/mois*100)+" %" : "—";
  document.getElementById("o-client").textContent= clients ? M(mois/clients)+" FCFA" : "—";
  const pct = v => mois ? v/mois*100 : 0;
  document.getElementById("m1").style.width = pct(recur)+"%";
  document.getElementById("m2").style.width = pct(web)+"%";
  document.getElementById("m3").style.width = pct(spot)+"%";
  document.getElementById("l1").textContent = "Abonnements "+Math.round(pct(recur))+" %";
  document.getElementById("l2").textContent = "Projets "+Math.round(pct(web))+" %";
  document.getElementById("l3").textContent = "Ponctuel "+Math.round(pct(spot))+" %";
}

}

}
